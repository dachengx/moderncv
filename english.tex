\section{Selected Research Experience}
\subsection{\href{https://xenonexperiment.org/}{XENONnT Dark Matter Project}}

\cventry{2022--2026}{Solar $^8$B CE$\nu$NS and Light Dark Matter Search}{}{}{}{Led solar neutrino analyses and developed signal and background models for light dark matter searches in the neutrino fog
\begin{itemize}
\item Led event selection, background modeling, and four-dimensional likelihood inference for XENONnT's first indication of solar $^8$B neutrino CE$\nu$NS and subsequent $3.3\sigma$ evidence from three science runs
\item Corresponding author of two PRL papers (2024, 2026); main contributor to XENONnT's first light dark matter search in the neutrino fog (PRL, 2025)
\end{itemize}}

\cventry{2024--2026}{Analysis Coordinator of XENONnT}{}{}{}{Elected to coordinate collaboration-wide physics analyses, serving from October 2024 to February 2026
\begin{itemize}
\item Chaired analysis meetings, led analysis reviews, and coordinated working-group priorities and deadlines
\item Coordinated shared data provisioning and analysis infrastructure; brought detector, calibration, background, and analysis teams together to reach common scientific decisions
\end{itemize}}

\cventry{2025--now}{Krypton Background Measurement System}{}{}{}{Developing XTRACT, a krypton-in-xenon measurement system targeting parts-per-quadrillion (ppq) sensitivity for rare-event background characterization
\begin{itemize}
\item Contribute to cryogenic design, construction, and commissioning of a cold-trap separation and residual-gas analysis system
\item Develop measurement automation and calibration to support krypton-background control for solar $pp$-neutrino and heavy WIMP searches
\end{itemize}}

\cventry{2024}{Convener of Computing and Analysis Tools Group}{}{}{}{Led the development of computing and analysis infrastructure supporting XENONnT physics analyses
\begin{itemize}
\item Developed and maintained HTC/HPC processing workflows involving 5\,PB of I/O
\item Designed a more stable data-reprocessing workflow adopted for subsequent XENONnT analyses
\end{itemize}}

\cventry{2023--2024}{Toy-MC Inference Framework}{}{}{}{Developed \href{https://github.com/XENONnT/alea/}{alea}, an open-source statistical inference framework for toy-Monte-Carlo confidence interval construction
\begin{itemize}
\item Designed modular inference workflows integrated with HPC computing for large-scale toy-Monte-Carlo studies
\item Adopted in XENONnT heavy WIMP, solar $^8$B neutrino, and $pp$-neutrino analyses, as well as by external collaborations
\end{itemize}}

\cventry{2022--2023}{GPU-based LXe Response Simulation and Inference}{}{}{}{Developed \href{https://github.com/XENONnT/appletree/}{appletree}, an open-source GPU framework for liquid-xenon detector response simulation and Bayesian inference
\begin{itemize}
\item Ported the simulation from CUDA/C++ to JAX/Python, improving usability and maintainability
\item Enabled MCMC-based detector response fitting subsequently used in XENONnT dark matter and neutrino analyses, including sub-keV light and charge yield calibration with a $^{88}$YBe photoneutron source
\end{itemize}}

\subsection{\href{http://jinping.hep.tsinghua.edu.cn/}{Jinping Neutrino Experiment\,(JNE)}}

\cventry{2019--2021}{PMT Waveform Analysis for Neutrino Detection}{}{}{}{Led a study of photon timing and charge reconstruction for neutrino and dark matter experiments, published in JINST (first author, 2022)
\begin{itemize}
\item Developed waveform reconstruction algorithms and a common framework to benchmark convolutional neural networks, deconvolution, and Bayesian methods
\item Improved photon counting and timing reconstruction, including charge resolution in low-occupancy regimes
\end{itemize}}

\subsection{\href{https://grid.tsinghua.edu.cn/site/}{Gamma-Ray Integrated Detectors\,(GRID)}}

\cventry{2018--2022}{CubeSat Gamma-ray Transient Events Analysis}{}{}{}{Developed a data processing pipeline for CubeSat gamma-ray burst\,(GRB) monitoring
