\cventry{2023--2025}{Convener of Computing and Analysis Tools Group}{}{}{}{Led development of computing and analysis software infrastructure
\begin{itemize}

\cventry{2023--2024}{Toy-MC Inference Framework}{}{}{}{Developed the \href{https://github.com/XENONnT/alea/}{open-source statistical inference framework} for toy-MC based confidence interval constructions
\begin{itemize}

\item Adopted by multiple analyses\,(heavy WIMP, solar $^8$B and $pp$ neutrinos, etc.) and external collaborations
\end{itemize}}

\cventry{2022--2023}{GPU-based LXe Response Simulation and Inference}{}{}{}{Developed a \href{https://github.com/XENONnT/appletree/}{open-source simulation and Bayesian inference framework} for liquid xenon detector response
\begin{itemize}
\item Rewrote CUDA\,(C++) framework into JAX\,(Python) for improved usability and maintainability
\item Enabled Bayesian inference with MCMC, widely used in dark matter and neutrino analyses in XENONnT
\end{itemize}}
